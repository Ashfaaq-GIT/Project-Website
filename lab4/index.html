#include <Arduino.h>
#include <ArduinoBLE.h>

void setup() {
  Serial.begin(115200);
  delay(2000); // give USB time
  Serial.println("BOOT");

  if (!BLE.begin()) {
    Serial.println("starting BLE failed!");
    while (1) { delay(10); }
  }

  Serial.print("BLE Address: ");
  Serial.println(BLE.address());

  BLE.setLocalName("Data Collection");
  BLE.advertise();
  Serial.println("Advertising...");
}

void loop() {
  // nothing
}

Step 3: Capture Training Data (Python + BLE)

Use the Python BLE receiver script to log accelerometer values into a CSV file on your PC. Then collect training data for punch and flex, repeating each gesture at least 10 times. :contentReference[oaicite:6]{index=6}

Windows Script (set BLE ADDRESS)

import asyncio
from datetime import datetime
from bleak import BleakClient

ADDRESS = "xx:xx:xx:xx:xx:xx"  # <-- Replace with your BLE Address
CHAR_UUID = "12345678-1234-5678-1234-56789abcdef1"
CSV_PATH = "accel_log.csv"

def int16_le(b0, b1):
    return int.from_bytes(bytes([b0, b1]), byteorder="little", signed=True)

def handle_notify(sender, data: bytearray):
    if len(data) != 8:
        return
    ctr = int16_le(data[0], data[1])
    ax_mg = int16_le(data[2], data[3])
    ay_mg = int16_le(data[4], data[5])
    az_mg = int16_le(data[6], data[7])

    ax_g = ax_mg / 1000.0
    ay_g = ay_mg / 1000.0
    az_g = az_mg / 1000.0

    ts = datetime.now().isoformat(timespec="milliseconds")
    line = f"{ts},{ctr},{ax_g:.3f},{ay_g:.3f},{az_g:.3f}"
    print(line)
    with open(CSV_PATH, "a", encoding="utf-8") as f:
        f.write(line + "\n")

async def main():
    print("Connecting to", ADDRESS)
    async with BleakClient(ADDRESS) as client:
        print("Connected:", client.is_connected)

        try:
            with open(CSV_PATH, "r", encoding="utf-8"):
                pass
        except FileNotFoundError:
            with open(CSV_PATH, "w", encoding="utf-8") as f:
                f.write("timestamp,ctr,ax_g,ay_g,az_g\n")

        await client.start_notify(CHAR_UUID, handle_notify)
        print("Streaming ACCEL... Ctrl+C to stop.")
        while True:
            await asyncio.sleep(1)

if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print("\nStopped.")
